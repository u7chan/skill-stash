スクロール演出24種のショールーム

演出を実際のスクロールで試し、CSS版 / Tailwind版のコードを確認してコピーできます。すべて静的ファイルで動作し、外部CDN・外部フォント・外部画像を読み込みません（オフライン可）。

  • ライブデモは常にCSS版です。Tailwindランタイムを読み込まないため、CSS / Tailwind切替はコード表示の切替とコピーとして実装しています。
  • カード内のiframeをスクロールすると演出が再生されます。ピン留め・横スクロールなど長いスクロールが必要な演出は、カード内の十分な高さのデモか「詳細デモを開く」で確認してください。
  • このページは catalog.json と examples/ を正本として scripts/build-showroom.py が生成しています。手編集は --check で検出されます。

Tailwind版は呼び出し元プロジェクトに Tailwind CSS が導入済みである前提のサンプルです。このショールームは Tailwind ランタイムを読み込まないため、ライブデモは CSS版を表示し、切替はコード表示とコピーに限定しています。ファイル: examples/tailwind/01-fade-in.html

実装の要点

  • 初期の非表示状態は @media (prefers-reduced-motion: no-preference) の中だけに置く
  • IntersectionObserver は一度検出したら unobserve して常時監視を避ける
  • transition は opacity と transform に限定し、レイアウトを動かさない

ブラウザ制約

  • IntersectionObserver はモダンブラウザ対応（Safari 12.1+）

prefers-reduced-motion

初期状態を作らないため、reduce 時は最初から表示される（no-preference でゲート）

アクセシビリティ

  • reduce 時に opacity: 0 が残らないようにする（本文が読めない事故を防ぐ）

正本: catalog.json / examples/css/01-fade-in.html / examples/tailwind/01-fade-in.html（Tailwind 導入済みプロジェクト向け・単体で開くとユーティリティは未適用）

  (function () {
    var section = document.querySelector('[data-sequence]');
    if (!section) { return; }
    var reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
    var frameElement = section.querySelector('[data-sequence-frame]');
    var counter = section.querySelector('[data-sequence-counter]');
    var total = 12;

    function frameSource(index) {
      var angle = (index / total) * 360;
      var svg = '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 320 200">'
        + '<rect width="320" height="200" fill="#0b1220"/>'
        + '<g transform="translate(160 100) rotate(' + angle.toFixed(1) + ')">'
        + '<rect x="-54" y="-54" width="108" height="108" rx="18" fill="#1d4ed8"/>'
        + '<rect x="-36" y="-36" width="72" height="72" rx="12" fill="#22d3ee"/>'
        + '</g>'
        + '<text x="12" y="188" fill="#8fa2c0" font-family="monospace" font-size="12">frame '
        + String(index + 1).padStart(2, '0') + ' / ' + total + '</text>'
        + '</svg>';
      return 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svg);
    }

    var frames = [];
    for (var index = 0; index < total; index += 1) {
      frames.push(frameSource(index));
    }

    var current = -1;
    function render(index) {
      if (index === current) { return; }
      current = index;
      frameElement.src = frames[index];
      counter.textContent = String(index + 1).padStart(2, '0') + ' / ' + total;
    }

    if (reduceMotion.matches) {
      render(Math.floor(total / 2)); /* reduce では代表フレームを1枚だけ表示する */
      return;

    function update() {
      ticking = false;

      var progress = span > 0 ? (-rect.top) / span : 0;
      progress = Math.max(0, Math.min(1, progress));
      render(Math.min(total - 1, Math.round(progress * (total - 1))));

実装の要点

  • position: sticky と親の高さだけで完結させ、JSを使わない
  • top の値は固定ヘッダーの高さと重ならないように決め、重なりを避ける
  • 親要素に overflow: hidden があると sticky が効かない点に注意する

ブラウザ制約

  • position: sticky は全モダンブラウザ対応（旧 Safari は -webkit- 接頭辞）

prefers-reduced-motion

動きを伴わないため常時固定のまま（reduce 専用の分岐は不要）

アクセシビリティ

  • sticky 内のリンクはキーボードで到達可能にする

  (function () {
    var reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
    var supportsTimeline = window.CSS && CSS.supports && CSS.supports('animation-timeline: view()');
    var section = document.querySelector('[data-pin-section]');
    if (!section || supportsTimeline || reduceMotion.matches) {
      return;
    }
    var steps = section.querySelectorAll('[data-pin-step]');
    var bar = section.querySelector('[data-pin-bar]');
    var total = steps.length;

    var ticking = false;
    function update() {
      ticking = false;
      var rect = section.getBoundingClientRect();
      var span = section.offsetHeight - window.innerHeight;
      var progress = span > 0 ? (-rect.top) / span : 0;
      progress = Math.max(0, Math.min(1, progress));
      var active = Math.min(total - 1, Math.floor(progress * total));
      steps.forEach(function (step, index) {
        step.classList.toggle('is-active', index === active);
      });
      if (bar) { bar.style.transform = 'scaleX(' + progress.toFixed(4) + ')'; }
    }
    window.addEventListener('scroll', function () {
      if (ticking) { return; }
      ticking = true;
      window.requestAnimationFrame(update);
    }, { passive: true });
    window.addEventListener('resize', update);
    update();
  })();

      ticking = false;
      var rect = section.getBoundingClientRect();
      var span = section.offsetHeight - window.innerHeight;
      var progress = span > 0 ? (-rect.top) / span : 0;
      progress = Math.max(0, Math.min(1, progress));
      var active = Math.min(total - 1, Math.floor(progress * total));
      steps.forEach(function (step, index) {

    }

      if (ticking) { return; }

  (function () {
    var section = document.querySelector('[data-hscroll-section]');
    var track = document.querySelector('[data-hscroll-track]');
    if (!section || !track) { return; }
    var reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
    var supportsTimeline = window.CSS && CSS.supports && CSS.supports('animation-timeline: view()');
    var travel = 0;
    var ticking = false;

    function update() {
      ticking = false;
      if (supportsTimeline || reduceMotion.matches) { return; }
      var rect = section.getBoundingClientRect();
      var span = section.offsetHeight - window.innerHeight;
      var progress = span > 0 ? (-rect.top) / span : 0;
      progress = Math.max(0, Math.min(1, progress));
      track.style.transform = 'translate3d(' + (-progress * travel).toFixed(1) + 'px, 0, 0)';
    }

    function measure() {
      var stage = section.querySelector('.hscroll__stage');
      var viewportWidth = stage ? stage.clientWidth : window.innerWidth;
      travel = Math.max(0, track.scrollWidth - viewportWidth);
      section.style.setProperty('--travel', travel.toFixed(0) + 'px'); /* CSS タイムライン用に距離を渡す */
      update();
    }

    window.addEventListener('scroll', function () {
      if (ticking) { return; }
      ticking = true;
      window.requestAnimationFrame(update);
    }, { passive: true });
    window.addEventListener('resize', measure);
    measure();

    var section = document.querySelector('[data-hscroll-section]');
    var track = document.querySelector('[data-hscroll-track]');
    if (!section || !track) { return; }
    var reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
    var supportsTimeline = window.CSS && CSS.supports && CSS.supports('animation-timeline: view()');
    var travel = 0;
    var ticking = false;

    function update() {
      ticking = false;
      if (supportsTimeline || reduceMotion.matches) { return; }
      var rect = section.getBoundingClientRect();
      var span = section.offsetHeight - window.innerHeight;
      var progress = span > 0 ? (-rect.top) / span : 0;
      progress = Math.max(0, Math.min(1, progress));
      track.style.transform = 'translate3d(' + (-progress * travel).toFixed(1) + 'px, 0, 0)';
    }

    function measure() {
      var stage = section.querySelector('.hscroll__stage');
      var viewportWidth = stage ? stage.clientWidth : window.innerWidth;
      travel = Math.max(0, track.scrollWidth - viewportWidth);
      section.style.setProperty('--travel', travel.toFixed(0) + 'px'); /* CSS タイムライン用に距離を渡す */
      update();
    }

    window.addEventListener('scroll', function () {
      if (ticking) { return; }
      ticking = true;
      window.requestAnimationFrame(update);
    }, { passive: true });
    window.addEventListener('resize', measure);
    measure();

  (function () {
    var links = document.querySelectorAll('[data-spy-link]');
    var sections = document.querySelectorAll('[data-spy-section]');
    if (!links.length || !sections.length) { return; }

    function activate(id) {
      links.forEach(function (link) {
        if (link.getAttribute('data-spy-link') === id) {
          link.setAttribute('aria-current', 'true');
        } else {
          link.removeAttribute('aria-current');
        }
      });
    }

    if (!('IntersectionObserver' in window)) {
      activate(sections[0].getAttribute('data-spy-section'));
      return;
    }
    /* ビューポート中央付近の細い帯に入ったセクションを現在地とする */
    var observer = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          activate(entry.target.getAttribute('data-spy-section'));
        }
      });
    }, { rootMargin: '-45% 0px -45% 0px', threshold: 0 });
    sections.forEach(function (section) { observer.observe(section); });
    activate(sections[0].getAttribute('data-spy-section'));
  })();
</script>

実装の要点

  • rootMargin でビューポート中央付近のセクションを検出対象にする
  • aria-current="true" を切り替え、視覚以外でも現在位置が分かるようにする
  • reduce では scroll-behavior: smooth を無効化する

ブラウザ制約

  • IntersectionObserver + CSS 変数（モダンブラウザ対応）

prefers-reduced-motion

スムーススクロールを無効化し、移動を即時反映にする
